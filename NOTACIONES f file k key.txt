DEFINICIÓN DE LA API
- "/login": GET: Usuario, Contraseña
- "/login": POST: usuario Contraseña (no todavía)
- "/files" GET: Usuario, Contraseña, Ruta
- "/file" GET: Usuario, Contraseña, Ruta
- "/file" POST: Usuario, Contraseña, Archivo,
{
	nombre,
	id,
	fecha,
	...
	clave,
	ruta,
	autorizaciones,
}
- "/file" PUT: Usuario, Contraseña, Archivo,
{
	nombre,
	id,
	fecha,
	...
	clave,
	ruta
	autorizaciones,
}
- "/file" DELETE: Usuario, Contraseña, Ruta
- "/folder" POST: Usuario, Contraseña, Ruta, Nombre
- "/folder" DELETE: Usuario, Contraseña, Ruta
- "/folder" POST: Usuario, Contraseña, Ruta, Nombre
